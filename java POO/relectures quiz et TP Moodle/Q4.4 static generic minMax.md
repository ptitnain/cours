class ArrayAlg
{
    public static <T extends Comparable<? super T>> Pair<T> minmax(T[] array)
    {
        if (array == null || array.length == 0)
        {
            return null;
        }

        T min = array[0];
        T max = array[0];
        for (int i = 1; i < array.length; i++)
        {
            if (array[i].compareTo(min) < 0)
            {
                min = array[i];
            }
            if (array[i].compareTo(max) > 0)
            {
                max = array[i];
            }
        }
        return new Pair<T>(min, max);
    }
}
