        // Create a LinkedList named colors and add the following colors in order.

        colors.add("Red");
        colors.add("Blue");
        colors.add("Green");
        colors.add("Yellow");
        colors.add("Purple");

        // Insert "White" at the beginning of the list.
        colors.addFirst("White");

        // Append "Black" at the end of the list.
        colors.addLast("Black");

        // Check if "Green" is present in the list.
        boolean containsGreen = colors.contains("Green");
        System.out.println("List contains Green: " + containsGreen);

        // Remove "Yellow" from the list.
        colors.remove("Yellow");

        // Print the final state of the colors list.
        System.out.println(colors);
    }
}
