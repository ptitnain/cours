public class ShopUtils {

    public static <T extends Sellable> T findLeastExpensiveItem(T[] items) {

        if (items == null || items.length == 0) {
            return null;
        }

        T cheapest = items[0];
        for (int i = 1; i < items.length; i++) {
            if (items[i].getPrice() < cheapest.getPrice()) {
                cheapest = items[i];
            }
        }
        return cheapest;
    }
}
