```
public class MapAPIExercise
{
   public static void main(String[] args)
   {
      Map<String, Employee> employeeRecords = new HashMap<>();

      // Adding employees to the map
      employeeRecords.put("101-12-3456", new Employee("John Doe"));
      employeeRecords.put("102-23-4567", new Employee("Jane Smith"));
      employeeRecords.put("103-34-5678", new Employee("Alice Johnson"));
      employeeRecords.put("104-45-6789", new Employee("Bob Brown"));

      // Print the entire map
      System.out.println(employeeRecords);

      // Remove the employee with ID "103-34-5678"
      employeeRecords.remove("103-34-5678");

      // Update the name of the employee with ID "102-23-4567" to "Jane Doe"
      employeeRecords.put("102-23-4567", new Employee("Jane Doe"));

      // Retrieve and print the Employee object associated with ID "104-45-6789"
      System.out.println(employeeRecords.get("104-45-6789"));

      // Iterate over the map and print each key-value pair
      employeeRecords.forEach((id, employee) -> System.out.println("ID: " + id + ", Employee: " + employee));
   }
}
```
